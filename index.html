class StepPolicy:
    attempts: int
    timeout: float
    side_effect: "none | idempotent | unsafe"

    def validate(self):
        if self.attempts > 1:
            require(self.timeout)
            reject(self.side_effect == "unsafe")

Resilient flow pattern
@flow(timeout_seconds=policy.deadline)
def ingest_market_data(date):
    symbols = guarded(fetch_symbols, policy.fetch)
    prices = parallel(fetch_prices, symbols)
    validate_coverage(prices)
    finalize_atomically(date, prices)
